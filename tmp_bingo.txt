import 'package:flutter/material.dart';

class BingoCard {
  final List<String> items;
  final Set<int> spottedItems = {};

  BingoCard({required this.items});

  void toggleSpot(int index) {
    if (spottedItems.contains(index)) {
      spottedItems.remove(index);
    } else {
      spottedItems.add(index);
    }
  }

  bool checkForBingo() {
    // Add center (FREE) space automatically
    final spots = {...spottedItems, 12};

    // Check rows
    for (var i = 0; i < 5; i++) {
      if (List.generate(5, (j) => i * 5 + j).every(spots.contains)) {
        return true;
      }
    }

    // Check columns
    for (var i = 0; i < 5; i++) {
      if (List.generate(5, (j) => i + j * 5).every(spots.contains)) {
        return true;
      }
    }

    // Check diagonals
    if (List.generate(5, (i) => i * 6).every(spots.contains)) {
      return true;
    }
    if (List.generate(5, (i) => (i + 1) * 4).every(spots.contains)) {
      return true;
    }

    return false;
  }
}

  @override
  void initState() {
    super.initState();
    _confettiController = AnimationController(
      duration: const Duration(seconds: 1),
      vsync: this,
    );

    // Generate a bingo card for each player
    for (var player in widget.players) {
      _playerCards[player] = _generateBingoCard();
    }
  }

  @override
  void dispose() {
    _confettiController.dispose();
    super.dispose();
  }

  // Generate a random bingo card (Australia)
  BingoCard _generateBingoCard() {
    final items = <String>[];
    final regionCategories = _regionCategories['Australia']!;
    final allItems = regionCategories.values.expand((items) => items).toList();

    // Shuffle and take first 24 items (center is FREE)
    allItems.shuffle();
    items.addAll(allItems.take(24));

    // Insert FREE space in the center (index 12)
    items.insert(12, 'FREE');

    return BingoCard(items: items);
  }

  void _toggleSpot(String player, int index) {
    final card = _playerCards[player]!;
    setState(() {
      card.toggleSpot(index);

      // Check for bingo after each spot
      if (card.checkForBingo()) {
        _showBingoAnimation(player);
      }
    });
  }

  void _showBingoAnimation(String player) {
    _confettiController.forward(from: 0);
    showDialog(
      context: context,
      builder: (context) => AlertDialog(
        title: Row(
          children: [
            const Icon(Icons.celebration, color: Colors.amber),
            const SizedBox(width: 8),
            Text('BINGO for $player!'),
          ],
        ),
        content: const Text('Congratulations! You got a BINGO!'),
        actions: [
          TextButton(
            onPressed: () => Navigator.pop(context),
            child: const Text('Continue Playing'),
          ),
          ElevatedButton(
            onPressed: () {
              setState(() {
                _playerCards[player] = _generateBingoCard();
              });
              Navigator.pop(context);
            },
            child: const Text('New Card'),
          ),
        ],
      ),
    );
  }

  Widget _buildBingoCard(String player, BingoCard card) {
    return Container(
      margin: const EdgeInsets.only(bottom: 24),
      decoration: BoxDecoration(
        color: Colors.white,
        borderRadius: BorderRadius.circular(16),
        boxShadow: [
          BoxShadow(
            color: Colors.black.withValues(alpha: 0.1),
            blurRadius: 6,
            offset: const Offset(0, 3),
          ),
        ],
      ),
      child: Column(
        children: [
          // Player header
          Container(
            width: double.infinity,
            padding: const EdgeInsets.all(12),
            decoration: BoxDecoration(
              color: Colors.pink,
              borderRadius: const BorderRadius.only(
                topLeft: Radius.circular(16),
                topRight: Radius.circular(16),
              ),
            ),
            child: Center(
              child: Text(
                player,
                style: const TextStyle(
                  fontSize: 18,
                  fontWeight: FontWeight.bold,
                  color: Colors.white,
                ),
              ),
            ),
          ),
          // Bingo grid
          GridView.builder(
            shrinkWrap: true,
            physics: const NeverScrollableScrollPhysics(),
            padding: const EdgeInsets.all(8),
            gridDelegate: const SliverGridDelegateWithFixedCrossAxisCount(
              crossAxisCount: 5,
              childAspectRatio: 1,
              crossAxisSpacing: 4,
              mainAxisSpacing: 4,
            ),
            itemCount: 25,
            itemBuilder: (context, index) {
              final item = card.items[index];
              final isSpotted = card.spottedItems.contains(index);
              final isFreeSpace = index == 12;

              return InkWell(
                onTap: isFreeSpace ? null : () => _toggleSpot(player, index),
                child: Container(
                  decoration: BoxDecoration(
                    color: isSpotted
                        ? Colors.pink.shade100
                        : isFreeSpace
                            ? Colors.pink.shade50
                            : Colors.white,
                    border: Border.all(
                      color: Colors.pink.shade200,
                      width: 1,
                    ),
                  ),
                  child: Center(
                    child: Padding(
                      padding: const EdgeInsets.all(4.0),
                      child: Text(
                        item,
                        textAlign: TextAlign.center,
                        style: TextStyle(
                          fontSize: 11,
                          fontWeight: isFreeSpace || isSpotted
                              ? FontWeight.bold
                              : FontWeight.normal,
                        ),
                      ),
                    ),
                  ),
                ),
              );
            },
          ),
        ],
      ),
    );
  }
